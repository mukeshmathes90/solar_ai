# Runtime dependencies for the microgrid monitoring platform.
#
#   pip install -r requirements.txt
#
# Flask serves the API and the dashboard from one process; numpy/pandas/
# scikit-learn/joblib run the machine-learning pipeline; matplotlib/seaborn
# render the report figures. pytest runs the test suite (see tests/).
#
# WHY THESE ARE RANGES AND NOT EXACT PINS
# ----------------------------------------
# scikit-learn model files are NOT portable across major scikit-learn
# releases: a .joblib trained with 1.3.2 can be *loaded* by 1.9 but then
# returns nonsense class probabilities, and 1.8 refuses to load it at all.
# The trained artifacts in ml/artifacts/ were produced with the stack below,
# so:
#
#   Python 3.8 - 3.11  ->  numpy 1.x + scikit-learn 1.3.x
#                       the shipped artifacts load as-is, no retraining needed.
#
#   Python 3.12 +      ->  numpy 2.x + scikit-learn 1.9.x
#                       the shipped artifacts are NOT usable, so run.bat
#                       detects that and retrains the models automatically
#                       on first start (about 2 minutes).
#
# The environment markers below let pip resolve the correct wheel set for
# whatever interpreter is running, which is why a hard pin such as
# numpy==1.24.4 must not be used here: it has no wheel for Python 3.12+ and
# pip then tries to build it from source, which fails.
#
# Supported: Python 3.8 - 3.14.

Flask>=2.3,<4
Flask-Cors>=4.0,<7
Werkzeug>=2.3,<4
Jinja2>=3.1,<4
joblib>=1.3

numpy>=1.24,<2; python_version < "3.12"
numpy>=2.0;      python_version >= "3.12"

pandas>=2.0,<3; python_version < "3.12"
pandas>=2.2;     python_version >= "3.12"

scikit-learn>=1.3,<1.4; python_version < "3.12"
scikit-learn>=1.9;       python_version >= "3.12"

scipy>=1.10,<1.12; python_version < "3.12"
scipy>=1.13;       python_version >= "3.12"

matplotlib>=3.7,<3.8; python_version < "3.12"
matplotlib>=3.9;       python_version >= "3.12"

seaborn>=0.13

pytest>=8
